179 RaceManager::MinorRaceModeType race_type,
180 const unsigned int number_of_players)
181 {
182 for (unsigned int n=0; n<UserConfigParams::m_saved_grand_prix_list.size(); n++)
183 {
184 SavedGrandPrix* gp = &UserConfigParams::m_saved_grand_prix_list[n];
185 if (gp->getGPID() == gpid &&
186 gp->getRaceType() == race_type &&
187 gp->getPlayerID() == player &&
188 gp->getPlayerKarts() == (int)number_of_players)
189 return gp;
190 }
191 return NULL;
