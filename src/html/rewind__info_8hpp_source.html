151 // ------------------------------------------------------------------------

154 // ------------------------------------------------------------------------
155 virtual bool isEvent() const { return true; }
156 // ------------------------------------------------------------------------
159 virtual void undo()
160 {
161 m_buffer->reset();
162 m_event_rewinder->undo(m_buffer);
163 } // undo
164 // ------------------------------------------------------------------------
168 virtual void replay()
169 {
170 // Make sure to reset the buffer so we read from the beginning
171 m_buffer->reset();
172 m_event_rewinder->rewind(m_buffer);
173 } // rewind
174 // ------------------------------------------------------------------------
176 BareNetworkString *getBuffer() { return m_buffer; }
177}; // class RewindIndoEvent
