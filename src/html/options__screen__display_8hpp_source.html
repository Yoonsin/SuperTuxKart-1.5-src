28struct Resolution
29{
30 int width;
31 int height;
32 bool fullscreen;
33
34 Resolution()
35 {
36 width = 0;
37 height = 0;
38 }
39
40 Resolution(int w, int h)
41 {
42 width = w;
43 height = h;
44 }
45
46 bool operator< (Resolution r) const
47 {
48 return width < r.width || (width == r.width && height < r.height);
49 }
50
51 float getRatio() const
52 {
53 return (float) width / height;
54 }
55};
