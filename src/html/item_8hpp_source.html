404 virtual bool hitKart(const Vec3 &xyz, const AbstractKart *kart=NULL) const
405 OVERRIDE
406 {
407 if (getPreviousOwner() == kart && getDeactivatedTicks() > 0)
408 return false;
409 Vec3 lc = quatRotate(getOriginalRotation(), xyz - getXYZ());
410 // Don't be too strict if the kart is a bit above the item
411 lc.setY(lc.getY() / 2.0f);
412 return lc.length2() < m_distance_2;
413 } // hitKart

415 bool rotating() const { return getType() != ITEM_BUBBLEGUM; }

436 return m_avoidance_points[1];
437 } // getAvoidancePoint
